def tg_send(text):
    if not TG_TOKEN:
        print("[DRY RUN]\n" + text)
        return
    try:
        requests.post("https://api.telegram.org/bot" + TG_TOKEN + "/sendMessage",
                      json={"chat_id": TG_CHAT, "text": text,
                            "disable_web_page_preview": True}, timeout=20)
    except Exception as e:
        print("TG send failed:", e)
